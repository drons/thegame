  ```js
  const MOBS_REF_RE = /^assets\/mobs\/\d{6}\.json$/;
  const MOBS_REF_CACHE = new Map();
  function resolveMobRef(file, ref, where) {
    if (typeof ref !== 'string' || !MOBS_REF_RE.test(ref)) {
      console.error(`${file}: ${where} «${ref}» — не ссылка на файл
        mobs (формат "assets/mobs/NNNNNN.json")`);
      process.exit(1);

    if (MOBS_REF_CACHE.has(ref)) return MOBS_REF_CACHE.get(ref);
    // readFileSync+JSON.parse в try/catch → exit 1 «файл отсутствует
    // или не JSON»; !tj || typeof tj.id !== 'string' || tj.id === ''
    // → exit 1 «id отсутствует»; успех → кэш + return tj.id
  ```
